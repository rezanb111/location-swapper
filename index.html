#!/usr/bin/env bash ############################################################################### # نصب MystNode روی سرور روسی با روت کردن ترافیک از طریق SOCKS5 خارجی # سرور روسی: 193.233.246.44 # پروکسی: socks5://bylex:bylex@147.45.77.71:1080 ############################################################################### set -euo pipefail IP="193.233.246.44" DATA_DIR="$HOME/mystnode" WG_PORTS="52000:52050" UI_PORT="4449" PROXY="socks5://bylex:bylex@altaria.proxy.rlwy.net:58542" echo "==> شروع نصب MystNode روی $IP" [[ $EUID -ne 0 ]] && echo "با root/sudo اجرا کنید" && exit 1 # --------------------------------------------------------------------------- # 1) هماهنگ‌سازی ساعت # --------------------------------------------------------------------------- echo "==> هماهنگ‌سازی ساعت سیستم" DEBIAN_FRONTEND=noninteractive apt-get update -qq DEBIAN_FRONTEND=noninteractive apt-get install -y -qq chrony curl systemctl enable --now chrony timedatectl set-ntp true # --------------------------------------------------------------------------- # 2) نصب Docker # --------------------------------------------------------------------------- if ! command -v docker &>/dev/null; then echo "==> نصب Docker" curl -fsSL https://get.docker.com | sh systemctl enable --now docker else echo "==> Docker از قبل نصب است" fi # --------------------------------------------------------------------------- # 3) تنظیم proxy برای Docker daemon # تمام ترافیک Docker (pull ایمیج + ترافیک داخل کانتینر) از پروکسی خارجی رد میشه # --------------------------------------------------------------------------- echo "==> تنظیم Docker proxy" mkdir -p /etc/systemd/system/docker.service.d cat > /etc/systemd/system/docker.service.d/proxy.conf << EOF [Service] Environment="HTTPS_PROXY=$PROXY" Environment="HTTP_PROXY=$PROXY" Environment="NO_PROXY=localhost,127.0.0.1,172.16.0.0/12,10.0.0.0/8" EOF # --------------------------------------------------------------------------- # 4) تنظیم proxy برای داخل کانتینرها (environment variables) # --------------------------------------------------------------------------- mkdir -p /etc/docker cat > /etc/docker/daemon.json << EOF { "proxies": { "http-proxy": "$PROXY", "https-proxy": "$PROXY", "no-proxy": "localhost,127.0.0.1,172.16.0.0/12,10.0.0.0/8" } } EOF systemctl daemon-reload systemctl restart docker sleep 3 # --------------------------------------------------------------------------- # 5) تست اتصال از طریق پروکسی # --------------------------------------------------------------------------- echo "==> تست اتصال به سرورهای Mysterium از طریق پروکسی" if docker run --rm curlimages/curl \ curl -s --max-time 15 \ https://observer.mysterium.network/api/v1/observed/hermes 2>/dev/null | grep -q "hermes\|address\|\["; then echo "==> اتصال موفق بود!" else echo "==> هشدار: تست اتصال پاسخ واضحی نداد، ادامه می‌دهیم..." fi # --------------------------------------------------------------------------- # 6) IP Forwarding # --------------------------------------------------------------------------- sysctl -w net.ipv4.ip_forward=1 grep -q '^net.ipv4.ip_forward' /etc/sysctl.conf \ && sed -i 's/^net.ipv4.ip_forward.*/net.ipv4.ip_forward=1/' /etc/sysctl.conf \ || echo 'net.ipv4.ip_forward=1' >> /etc/sysctl.conf # --------------------------------------------------------------------------- # 7) پوشه داده # --------------------------------------------------------------------------- mkdir -p "$DATA_DIR" # --------------------------------------------------------------------------- # 8) فایروال # --------------------------------------------------------------------------- if command -v ufw &>/dev/null && ufw status | grep -q "Status: active"; then ufw allow "$UI_PORT/tcp" ufw allow 52000:52050/udp fi # --------------------------------------------------------------------------- # 9) دانلود ایمیج # --------------------------------------------------------------------------- echo "==> دانلود ایمیج MystNode" docker pull mysteriumnetwork/myst:latest # --------------------------------------------------------------------------- # 10) اجرای نود # کانتینر با متغیرهای proxy اجرا میشه تا ترافیک داخلیش هم از پروکسی رد بشه # --------------------------------------------------------------------------- echo "==> اجرای نود" docker rm -f myst-node 2>/dev/null || true docker run -d \ --name myst-node \ --restart unless-stopped \ --cap-add NET_ADMIN \ --device=/dev/net/tun \ -e HTTPS_PROXY="$PROXY" \ -e HTTP_PROXY="$PROXY" \ -e https_proxy="$PROXY" \ -e http_proxy="$PROXY" \ -e NO_PROXY="localhost,127.0.0.1,172.16.0.0/12,10.0.0.0/8" \ -v "$DATA_DIR:/var/lib/mysterium-node" \ -p "$IP:$UI_PORT:4449" \ -p "$IP:52000-52050:52000-52050/udp" \ mysteriumnetwork/myst:latest \ service --agreed-terms-and-conditions --wireguard.listen.ports "$WG_PORTS" wireguard sleep 8 docker ps --filter "name=myst-node" echo "" echo "==================================================================" echo " نصب تمام شد!" echo " ۳۰ ثانیه صبر کنید بعد پنل رو باز کنید:" echo " پنل: http://$IP:$UI_PORT" echo "" echo " مراحل بعدی در پنل:" echo " 1) Register identity (پرداخت ~1 USDC روی Polygon)" echo " 2) Settings > Mystnodes.com Integrations" echo " API key: GKRVndEnkWBXLgKOLzvDkYZ8xtnwHMSIePnlPW1k" echo "" echo " چک لاگ: docker logs -f myst-node" echo "=================================================================="
